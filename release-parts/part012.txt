function archiumArtifactAnomalyMap(artifact){
 const result={'ТЧ':[],'ЧН':[],'ЗП':[]};
 const key=normalize(artifact.name);
 for(const anomaly of DB){
  if(anomaly.cat!=='anomalies'||!anomaly.anomalyByGame)continue;
  for(const g of ['ТЧ','ЧН','ЗП']){
   const record=anomaly.anomalyByGame[g];
   if(record && (record.artifacts||[]).some(name=>normalize(name)===key))result[g].push(anomaly);
  }
 }
 return result;
}
